cmake_minimum_required(VERSION 3.8)
project(ocp_planner)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
set(CMAKE_CXX_FLAGS_RELEASE "-O3")

# Find dependencies for ROS2
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclpy REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(rosidl_default_generators REQUIRED)

find_package(OpenCV REQUIRED)
find_package(Eigen3 REQUIRED)

################################################
## Declare ROS messages and services ##
################################################

# Generate messages
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/DynaObstacle.msg"
  "msg/ObstacleArray.msg"
  "msg/RobotState.msg"
  "msg/HumanState.msg"
  "msg/ObstacleState.msg"
  "msg/WallState.msg"
  "msg/Point.msg"
  "msg/PolyState.msg"
  "msg/JointState.msg"
  "msg/ControlVar.msg"
  "srv/OcpLocalPlann.srv"
  DEPENDENCIES std_msgs geometry_msgs nav_msgs
)

################################################
## Build ##
################################################

include_directories(
  ${PROJECT_SOURCE_DIR}
  ${OpenCV_INCLUDE_DIRS}
  ${EIGEN3_INCLUDE_DIR}
)

# ROS2 node executable
add_executable(${PROJECT_NAME}_node
  src/main.cc
  src/astar.cc
  src/smooth.cc
  src/lookahead.cc
  src/planner.cc
  src/mpc.cc
)

ament_target_dependencies(${PROJECT_NAME}_node
  rclcpp
  std_msgs
  geometry_msgs
  nav_msgs
)

target_link_libraries(${PROJECT_NAME}_node
  ${OpenCV_LIBRARIES}
  casadi
  dw
)

################################################
################# pybind11 #####################
################################################

# IMPORTANT: Update these paths to your conda environment
set(PYTHON_EXECUTABLE "$ENV{CONDA_PREFIX}/bin/python")
set(PYTHON_INCLUDE_DIRECTORY "$ENV{CONDA_PREFIX}/include/python3.8")

# Add pybind11 subdirectory
add_subdirectory(extern/pybind11)

# Build Python binding module (STANDALONE - no ROS dependency)
pybind11_add_module(${PROJECT_NAME}_py
  src/ocp_bind.cc
  src/astar.cc
  src/smooth.cc
  src/lookahead.cc
  src/planner.cc
  src/mpc.cc
)

target_link_libraries(${PROJECT_NAME}_py
  PRIVATE
  ${OpenCV_LIBRARIES}
  casadi
)

################################################
## Install ##
################################################

install(TARGETS ${PROJECT_NAME}_node
  DESTINATION lib/${PROJECT_NAME}
)

# Install Python binding to lib directory
install(TARGETS ${PROJECT_NAME}_py
  LIBRARY DESTINATION lib
)

ament_package()
